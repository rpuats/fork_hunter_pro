                let callbackSubmitButtonDisabledTimeStamp = sessionStorage.getItem('callbackSubmitButtonDisabledTimeStamp' + accountId);
                if (callbackSubmitButtonDisabledTimeStamp) {
                    let timeDiff = (parseInt(callbackSubmitButtonDisabledTimeStamp) + retryUnavailableInSeconds * 1000) - Date.now();
                    let timerSecondsLeft = timeDiff / 1000;
                    if (timeDiff > 0) {
                        retryUnavailableHandler(timerSecondsLeft, accountId);
                    }
                }
            }
        });
